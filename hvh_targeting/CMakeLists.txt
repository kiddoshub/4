cmake_minimum_required(VERSION 3.16)
project(hvh_targeting VERSION 2.0 LANGUAGES CXX)

add_library(targeting_math INTERFACE)
target_include_directories(targeting_math INTERFACE src)
target_compile_features(targeting_math INTERFACE cxx_std_17)

include(CTest)
if(BUILD_TESTING)
    add_executable(targeting_math_test tests/targeting_math_test.cpp)
    target_link_libraries(targeting_math_test PRIVATE targeting_math)
    add_test(NAME targeting_math COMMAND targeting_math_test)
endif()

if(WIN32)
    add_executable(hvh_targeting src/main.cpp)
    configure_file(hvh_targeting.ini ${CMAKE_CURRENT_BINARY_DIR}/hvh_targeting.ini COPYONLY)
    target_link_libraries(hvh_targeting PRIVATE targeting_math user32)
    if(MSVC)
        target_compile_options(hvh_targeting PRIVATE /W4)
    else()
        target_compile_options(hvh_targeting PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(MINGW)
        target_link_options(hvh_targeting PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
endif()
